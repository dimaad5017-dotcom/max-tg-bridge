# Верхняя граница у каждой строки — не придирчивость, а страховка от тихой поломки.
# Установка запускается заново при каждом обновлении моста, и pip берёт самое свежее.
# Выйди завтра aiogram 4.0 с переписанным API — мост упал бы прямо при импорте, а
# человек увидел бы только «мост падает сразу после запуска» и ни слова о причине.
#
# Но и намертво прибивать версии нельзя: MAX меняет свой протокол, и починки к нему
# приезжают как раз обновлением maxapi-python. Поэтому не «ровно эта», а «любая, пока
# не сменилась старшая цифра»: внутри одной старшей версии ломать API не принято.
maxapi-python>=2.4,<3
aiogram>=3.15,<4
python-dotenv>=1.0,<2

# напрямую. До сих пор она приезжала прицепом к aiogram, и всё держалось на том, что
# aiogram её тянет. Перестанет тянуть — мост не запустится, а по списку будет не понять
# почему: того, чего в нём нет, никто и не ищет.
aiohttp>=3.9,<4
